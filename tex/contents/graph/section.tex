\cpp[4]{SCC}{scc.hpp}[
	\fld{Use} \texttt{vector<vector<int>> g(n + 1); g[u].push\_back(v); SCC scc(g);}
	\fld{Result} \texttt{scc.cnt} components; \texttt{scc.comp[u]} is in $1..\texttt{cnt}$.
	An edge $u \to v$ has $\texttt{comp[u]} \ge \texttt{comp[v]}$: ids $\texttt{cnt}, \dots, 1$ are a topological order.

]\label{scc}
\cpp[4]{2-SAT}{2sat.hpp}[
	\fld{Use} \texttt{TwoSat ts(n); ts.either(a, b); bool ok = ts.solve();}
	\fld{Result} \texttt{solve()} says whether all clauses hold at once; then \texttt{val[x]} is a satisfying value of $x \in 1..n$.
	\fld{Needs} literals $\pm x$ with $1 \le x \le n$. Uses SCC (\S\ref{scc}).
	\fld{either} \texttt{either(a, b)} adds $a \lor b$. Implication $a \Rightarrow b$ is \texttt{either(-a, b)}; forcing $x$ is \texttt{either(x, x)}.
]\label{2sat}
\cpp[4]{Lowlink: bridges, articulation points, BCC}{lowlink.hpp}[
	\fld{Use} \texttt{g[u].push\_back(\{v, id\}); g[v].push\_back(\{u, id\}); LowLink low(g, m);}
	\fld{Result} \texttt{isBridge[id]}; \texttt{isCut[u]}; \texttt{bcc[id]} $\in 1..\texttt{bccCnt}$ is the biconnected component of edge $id$.
	A BCC's vertices are the endpoints of its edges; 2-edge-connected components are a flood fill that never crosses a bridge.
	\fld{Needs} distinct ids in $0..m$. Multi-edges are never bridges; a self-loop gets \texttt{bcc} $= 0$.
]\label{lowlink}
\cpp[4]{Euler path / circuit}{euler.hpp}[
	\fld{Use} \texttt{auto w = eulerWalk(g, m, s);} with \texttt{g[u]} holding \texttt{\{v, id\}}.
	\fld{Result} \texttt{w[i].fi} is the $i$-th vertex and \texttt{w[i].se} the edge into it (\texttt{-1} for $i = 0$). Empty if no walk from $s$ uses all $m$ edges.
	\fld{Needs} distinct ids in $0..m$. An undirected self-loop may be listed once or twice.
	\fld{Start} $s$ is an odd-degree vertex (undirected) or one with out $-$ in $= 1$ (directed); else any vertex with an edge, any $s$ if $m = 0$.

\subsection{Paths \& spanning trees}
\cpp[4]{SPFA}{spfa.hpp}[
	\fld{Use} \texttt{vector<vector<pair<int, ll>>> g(n + 1); SPFA sp(g, s);}
	\fld{Result} \texttt{dist[v]} is \texttt{oo} if unreachable, \texttt{-oo} if a walk $s \leadsto v$ crosses a negative cycle. \texttt{par[v]} precedes $v$ (0 at $s$), valid for $|\texttt{dist[v]}| < \texttt{oo}$.
	\fld{Needs} $n \cdot \max |w| < 10^{18}$. No $O(VE)$ bound: crafted inputs are exponential. Shuffle each \texttt{g[u]} once.
	\fld{negCycle} flags a negative cycle reachable from $s$. To detect one anywhere, use \texttt{g(n + 2)}; start at $n + 1$ with 0-edges to all vertices.
]\label{spfa}
\topic{Print a negative cycle}
Use it when the cycle itself is wanted: \texttt{par} from SPFA need not contain it yet.
\begin{facts}
	\item Run Bellman--Ford: $n$ passes over all edges, keeping \texttt{par}.
	\item Take any vertex $x$ relaxed in pass $n$.
	\item Set $x \gets \texttt{par[}x\texttt{]}$ $n$ times; now $x$ lies on a negative cycle.
	\item Walk \texttt{par} from $x$ until $x$ repeats; reverse the list for the edge direction.
\end{facts}
\topic{Min mean cycle}
Use it to minimise the average weight $\bar w(C)$ of a cycle $C$, when the graph has a cycle. Fix an integer scale $K \ge 1$ (e.g.\ $K = n$) and an integer $\lambda$:
\[ \bar w(C) < \tfrac{\lambda}{K} \iff \textstyle\sum_{e \in C} (K w_e - \lambda) < 0 \]
\begin{facts}
	\item Give every edge the weight $K w_e - \lambda$ and look for a negative cycle anywhere (the \texttt{negCycle} line above).
	\item Binary search the largest $\lambda$ in $[-KW, KW]$ with no negative cycle, where $W = \max |w_e|$; the minimum mean lies in $[\lambda / K, (\lambda + 1) / K)$.
	\item Overflow: needs $n (K W + |\lambda|) < 10^{18}$.
	\item The exact rational value: Karp's algorithm, $O(VE)$ (not printed).
\end{facts}
\cpp[4]{Directed MST}{dmst.hpp}[
	\fld{Use} \texttt{vector<vector<pair<int, ll>>> g(n + 1); DirectedMST dm(g, r);}
	\fld{Result} \texttt{dm.cost}, the weight of a minimum rooted arborescence (reaches every vertex); \texttt{dm.par[v]} is the tail of $v$'s edge, \texttt{par[r]} $= 0$.
	\fld{Needs} $2n \cdot \max |w| < 9 \cdot 10^{18}$. \texttt{cost == LLONG\_MAX}: some vertex is unreachable. Self-loops are skipped. Uses Rollback DSU (\S\ref{rollbackdsu}).

\cpp[4]{Manhattan MST}{manhattanmst.hpp}[
	\fld{Use} \texttt{vector<pll> p(n + 1); auto e = manhattanMST(p);}
	\fld{Result} $n - 1$ edges \texttt{\{w, u, v\}} of a minimum spanning tree in increasing $w$, where $w = |x_u - x_v| + |y_u - y_v|$.
	\fld{Needs} $|x|, |y| \le 10^{17}$. Chebyshev distance: rotate to $(x + y, x - y)$ first, then halve each $w$.
]\label{manhattanmst}
\entry{Matrix-tree theorem}\label{matrixtree}
{\small\tightdisplay
\topic{Undirected}
Use it to count the spanning trees of a multigraph on vertices $1..n$.
\[ L = D - A, \qquad t(G) = \det L_{\hat r \hat r} \]
$D$ is the diagonal of degrees. $A[u][v]$ is the number of edges between $u$ and $v$.
$L_{\hat r \hat r}$ is $L$ without row $r$ and column $r$, for any $r$. Self-loops are ignored.
\par Example: $K_n$ has $n^{n-2}$ spanning trees, so $K_4$ has 16.
\topic{Computing the determinant}
Use Gauss--Jordan (\S\ref{gauss}) to get the count modulo a prime.
\begin{facts}
	\item Let \texttt{a} be \texttt{vector<vector<Mint<MOD>{}>{}>} of size $(n - 1) \times (n - 1)$.
	\item Fill it with $L$, skipping row $r$ and column $r$. It is 0-based.
	\item The count is \texttt{gauss(a, n - 1).se}, in $O(n^3)$. For $n = 1$ it gives 1.
	\item The exact count can be huge (up to $n^{n-2}$ for simple graphs): only its value modulo the prime is safe.
\end{facts}
\topic{Directed, rooted at $r$}
Use it to count the spanning arborescences rooted at $r$.
\begin{facts}
	\item Edges pointing away from $r$ (every $v \ne r$ has one edge in):
	\[ L = D_{in} - A. \]
	\item Edges pointing toward $r$ (every $v \ne r$ has one edge out):
	\[ L = D_{out} - A. \]
	\item $A[u][v]$ is the number of edges $u \to v$. Drop row $r$ and column $r$ as above.
\end{facts}
Example: root 1, edges $1 \to 2$, $1 \to 3$, $2 \to 3$ (away from 1): vertex 3 takes its edge from 1 or from 2, so there are 2 arborescences.
\topic{Weighted}
Use it to sum, over all trees, the product of the edge weights.
\begin{facts}
	\item Add the weight instead of 1 to $A$ and $D$, then take the same determinant.
\end{facts}
Example: triangle with weights 1, 2, 3: the trees give $1 \cdot 2 + 1 \cdot 3 + 2 \cdot 3 = 11$.
}

\subsection{Trees}
\cpp[4]{HLD}{hld.hpp}[
	\fld{Use} \texttt{HLD hld(g); for(auto [l, r]: hld.path(u, v)) s += seg.get(l, r).sum;}
	\fld{Result} \texttt{par}, \texttt{depth}, \texttt{head}, \texttt{sub} (subtree size); \texttt{par[root]} $= 0$.
	\fld{Needs} \texttt{g} is an undirected tree, each edge in both lists; \texttt{u}, \texttt{v} in \texttt{root}'s component. Fill \texttt{seg} from \texttt{a[hld.pos[u]] = \{value, 1\}} (\S\ref{lazyseg}).
	\fld{lca} \texttt{lca(u, v)} is the lowest common ancestor.
	\fld{path} covers $u \leadsto v$ by closed \texttt{pos} ranges, unordered: operations must commute. \texttt{edge = true} stores $(\texttt{par[v]}, v)$ at $v$, omitting the LCA.
]\label{hld}
\cpp[4]{Virtual tree}{virtualtree.hpp}[
	\fld{Use} \texttt{HLD hld(g); auto t = virtualTree(hld, vs);}
	\fld{Result} \texttt{t} holds pairs \texttt{\{v, p\}} in preorder over \texttt{vs} and their lowest common ancestors; $p$ is $v$'s parent, 0 for the first.
	\fld{Needs} \texttt{hld} is built on the same tree (\S\ref{hld}); \texttt{vs} may repeat vertices or be empty.
]\label{virtualtree}
\cpp[4]{Centroid decomposition}{centroid.hpp}[
	\fld{Use} \texttt{Centroid cd(g);} then, for each $c$, a DFS from $c$ that enters only $v$ with \texttt{cd.level[v] > cd.level[c]}.
	\fld{Result} that DFS visits the component of $c$. A vertex lies in $\le \log_2 n + 1$ components, so linear work in each totals $O(n \log n)$.
	\fld{Needs} a forest, each edge in both lists.
]\label{centroid}
