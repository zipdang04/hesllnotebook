\subsection{Connectivity}
\cpp[4]{SCC + 2-SAT}{scc.hpp}[
	Component ids are in topological order (edge $u\to v$ $\Rightarrow$ \texttt{comp[u]} $\le$ \texttt{comp[v]}). Iterative, safe at $10^6$ vertices. Force $x$ true: \texttt{either(x, x)}; $x \Rightarrow y$: \texttt{either(\textasciitilde x, y)}.
]
\cpp[4]{Lowlink: bridges, articulation points, BCC, 2ECC}{lowlink.hpp}[
	Self-loops and multi-edges OK (parent skipped by edge id). Recursive: depth up to $n$ (needs a big stack for $n \approx 5\cdot10^5$). Block-cut tree: join each block node to its vertices.
]
\cpp[4]{Euler path / circuit}{euler.hpp}[
	Circuit exists iff \texttt{solve()} and \texttt{vs[0] == vs.back()}. \texttt{false} also when edges lie in several components (isolated vertices are fine).

\subsection{Paths \& spanning trees}
\cpp[4]{SPFA}{spfa.hpp}[
	Path: walk \texttt{par} back from $t$ (edge ids, so keep each edge's tail). Other \texttt{adj}: filter edges (\texttt{if (w <= lim)}), change weights (min mean cycle: $w - \lambda$), implicit graphs (grids).
]
