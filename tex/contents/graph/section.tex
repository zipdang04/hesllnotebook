\entry{Matrix-tree theorem}
\small\tightdisplay
\topic{Undirected}
The number of spanning trees is the determinant of the Laplacian with any row and column $i$ removed (loops ignored):
\[ L = D - A, \qquad A_{uv} = \#\text{edges } uv \]
\topic{Directed}
Remove row and column $r$; $A_{uv} = \#(u \to v)$.
\begin{facts}
\item Out-trees (edges away from $r$): $L = D_{\text{in}} - A$.
\item In-trees: $L = D_{\text{out}} - A$.
\end{facts}
\topic{Weighted}
Put weights in $A$ and $D$ to get the sum over trees:
\[ \sum_T \prod_{e \in T} w_e \]
Determinant mod $p$: Gauss--Jordan (\S6.2.1).

\subsection{Trees}
\cpp[4]{HLD}{hld.hpp}[
	\fld{Needs} connected tree; \texttt{build()} once. Iterative (fine for $n = 5\cdot10^5$).
	\fld{\texttt{f(l, r)}} half-open \texttt{pos} ranges in arbitrary order, direction lost: commutative aggregates only.
	\fld{Edges} edge $(par_c, c)$ is stored at \texttt{pos[c]}.
	\fld{Distance}
	\[ d(u,v) = dep_u + dep_v - 2\,dep_{lca} \]
]
\cpp[4]{Centroid decomposition}{centroid.hpp}[
	\fld{Note} forest OK. Only centroid recursion (depth $\le \log_2 n + 1$); sizes via BFS, safe at $n = 2\cdot10^5$ (path) on a default stack.
	\fld{In \texttt{f(c)}} \texttt{bfs(c, -1)} = whole piece; \texttt{bfs(v, c)} per non-removed neighbour $v$ = one subtree; \texttt{dep} = distance to $c$.
]
\cpp[4]{Virtual tree}{virtualtree.hpp}[
	\fld{Note} duplicates in $S$ OK; $|vs| \le 2|S|-1$.
	\fld{Order} \texttt{pa[i] < i}: reverse index order is bottom-up for DP.
	\fld{\texttt{id}} $O(n)$ once, reused without reset.
	\fld{Weighted} lengths: use weighted depths instead of \texttt{dep}.
