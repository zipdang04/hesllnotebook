\subsection{Connectivity}
\cpp[4]{SCC}{scc.hpp}[
	\fld{Use} \texttt{vector<vector<int>> g(n + 1); g[u].push\_back(v); SCC scc(g);}
	\fld{Result} \texttt{scc.cnt} components; \texttt{scc.comp[u]} is in $1..\texttt{cnt}$.
	An edge $u \to v$ has $\texttt{comp[u]} \ge \texttt{comp[v]}$: ids $\texttt{cnt}, \dots, 1$ are a topological order.
	\fld{Needs} nothing: self-loops, multi-edges and isolated vertices are fine.
]\label{scc}
\cpp[4]{2-SAT}{2sat.hpp}[
	\fld{Use} \texttt{TwoSat ts(n); ts.either(a, b); if(ts.solve()) .. ts.val[x]}
	\fld{Result} \texttt{solve()} says whether all clauses hold at once; then \texttt{val[x]} is a satisfying value of $x \in 1..n$.
	\fld{Needs} literals $\pm x$ with $1 \le x \le n$. Uses SCC (\S\ref{scc}).
	\fld{either} \texttt{either(a, b)} adds $a \lor b$. Implication $a \Rightarrow b$ is \texttt{either(-a, b)}; forcing $x$ is \texttt{either(x, x)}.
]\label{2sat}
\cpp[4]{Lowlink: bridges, articulation points, BCC}{lowlink.hpp}[
	\fld{Use} \texttt{g[u].push\_back(\{v, id\}); g[v].push\_back(\{u, id\}); LowLink low(g, m);}
	\fld{Result} \texttt{isBridge[id]}; \texttt{isCut[u]}; \texttt{bcc[id]} $\in 1..\texttt{bccCnt}$ is the biconnected component of edge $id$.
	A BCC's vertices are the endpoints of its edges; 2-edge-connected components are a flood fill that never crosses a bridge.
	\fld{Needs} distinct ids in $0..m$. Multi-edges are never bridges; a self-loop gets \texttt{bcc} $= 0$.
]\label{lowlink}
\cpp[4]{Euler path / circuit}{euler.hpp}[
	\fld{Use} \texttt{auto w = eulerWalk(g, m, s);} with \texttt{g[u]} holding \texttt{\{v, id\}}.
	\fld{Result} \texttt{w[i].fi} is the $i$-th vertex and \texttt{w[i].se} the edge into it (\texttt{-1} for $i = 0$). Empty if no walk from $s$ uses all $m$ edges.
	\fld{Needs} distinct ids in $0..m$. Start $s$: undirected, an odd-degree vertex; directed, one with out $-$ in $= 1$; else any vertex with an edge.
]\label{euler}

\subsection{Paths \& spanning trees}
\cpp[4]{SPFA}{spfa.hpp}[
	\fld{Use} \texttt{vector<vector<pair<int, ll>>> g(n + 1); SPFA sp(g, s);}
	\fld{Result} \texttt{dist[v]} is \texttt{oo} if unreachable, \texttt{-oo} if a walk $s \leadsto v$ crosses a negative cycle. \texttt{par[v]} precedes $v$ (0 at $s$), valid for $|\texttt{dist[v]}| < \texttt{oo}$.
	\fld{Needs} $n \cdot \max |w| < 10^{18}$. Shuffle each \texttt{g[u]} once against anti-SPFA tests.
	\fld{negCycle} flags a negative cycle reachable from $s$. To detect one anywhere, use \texttt{g(n + 2)}; start at $n + 1$ with 0-edges to all vertices.
]\label{spfa}
\topic{Print a negative cycle}
Use it when the cycle itself is wanted: \texttt{par} from SPFA need not contain it yet.
\begin{facts}
	\item Run Bellman--Ford: $n$ passes over all edges, keeping \texttt{par}.
	\item Take any vertex $x$ relaxed in pass $n$.
	\item Set $x \gets \texttt{par[}x\texttt{]}$ $n$ times; now $x$ lies on a negative cycle.
	\item Walk \texttt{par} from $x$ until $x$ repeats; reverse the list for the edge direction.
\end{facts}
\topic{Min mean cycle}
Use it to minimise the average weight $\bar w(C)$ of a cycle $C$. Fix an integer scale $K$ (e.g.\ $K = n$) and an integer $\lambda$:
\[ \bar w(C) < \tfrac{\lambda}{K} \iff \textstyle\sum_{e \in C} (K w_e - \lambda) < 0 \]
\begin{facts}
	\item Give every edge the weight $K w_e - \lambda$ and look for a negative cycle anywhere (the \texttt{negCycle} line above).
	\item Binary search the largest $\lambda$ in $[-KW, KW]$ with no negative cycle, where $W = \max |w_e|$; the minimum mean lies in $[\lambda / K, (\lambda + 1) / K)$.
	\item Overflow: needs $n (K W + |\lambda|) < 10^{18}$.
	\item The exact rational value: Karp's algorithm, $O(VE)$ (not printed).
\end{facts}
