\cpp[4]{Z-function}{zfunc.hpp}[
	\fld{Use} \texttt{vector<int> z = zFunc(s);}
	\fld{Result} \texttt{zFunc(p + '\#' + s)[i] == m} means $p$ (length $m$) starts at \texttt{s[i - m - 1]}.
	A $0 < d < n$ is a period of $s$ exactly when \texttt{d + z[d] == n}.
	\fld{Needs} a separator \texttt{'\#'} that occurs in neither string.
]\label{zfunc}
\cpp[4]{Rolling hash}{strhash.hpp}[
	\fld{Use} \texttt{StrHash h(s); if(h.get(l1, r1) == h.get(l2, r2))}
	\fld{Result} \texttt{get(l, r)} hashes $s_l \dots s_r$; \texttt{get(l, l - 1)} is 0, the empty string.
	All objects share the base, so hashes of two strings compare.
	A 0-based start $p$ and length $L$ is the range \texttt{get(p + 1, p + L)}.
	\fld{Needs} character codes 1 to 127; a collision has probability about $n / 2^{61}$ per comparison.
	\fld{getRev} returns the hash of $s_r s_{r-1} \dots s_l$; $s_l \dots s_r$ is a palindrome with high probability when \texttt{getRev(l, r) == get(l, r)}.
]\label{strhash}
{\footnotesize\tightdisplay
\topic{LCP, order and period of substrings}
To compare two substrings with 0-based starts $i$, $j$:
\begin{facts}
	\item Binary search the largest $L \le \min(\ell_i, \ell_j)$ ($\ell_i$ the length of the substring from $i$; $n - \max(i, j)$ for suffixes) with \texttt{get(i + 1, i + L) == get(j + 1, j + L)}: the common prefix.
	\item If a character is left in both, $s_{i+L}$ against $s_{j+L}$ (0-based) decides the order; otherwise the shorter one is smaller.
	\item $d$ is a period of $s_l \dots s_r$ when \texttt{get(l, r - d) == get(l + d, r)}.
\end{facts}
}
\cpp[4]{Suffix array + LCP}{suffixarray.hpp}[
	\fld{Use} \texttt{SuffixArray sa(s);} for rotations; \texttt{SuffixArray sa(s + '\$');} for suffixes.
	\fld{Result} \texttt{sa.sa[i]} is rotation rank $i$'s start; \texttt{sa.rnk} is its inverse.
	\texttt{sa.lcp[i]} is ranks $i-1$, $i$'s common prefix length ($\le n$); \texttt{lcp[0]} is 0.
	Ties use increasing starts; \texttt{sa[0]} is the first minimum rotation.
	\fld{Needs} For suffixes, final \texttt{'\$'} is unique and smallest; \texttt{sa[0]} is $n$, and ranks $1..n$ are the suffixes of $s$.
	LCS of $s, t$: \texttt{s + '\$' + t + '\#'}, both absent from $s, t$ and \texttt{'\#'} $<$ \texttt{'\$'} $<$ data.
]\label{suffixarray}
\cpp[4]{Aho--Corasick}{ahocorasick.hpp}[
	\fld{Use} \texttt{Aho a; for(auto \&p: ps) e.push\_back(a.add(p)); a.build();}
	\fld{Result} \texttt{t[u].nxt[c]} is the goto table; \texttt{t[u].cnt} counts pattern suffixes of $u$.
	Scan: \texttt{hit[u]++} at each node $u$.
	In reverse BFS \texttt{order}, add \texttt{hit[u]} to \texttt{hit[t[u].link]}; pattern $i$ occurs \texttt{hit[e[i]]} times.
	\fld{Needs} Non-empty patterns over \texttt{FIRST}..\texttt{FIRST + K - 1}; all \texttt{add} calls before \texttt{build}.
	\fld{countMatch} Counts all pattern occurrences in \texttt{s}.
]\label{ahocorasick}
\cpp[4]{Suffix automaton}{suffixautomaton.hpp}[
	\fld{Use} \texttt{SAM sam; for(char c: s) sam.add(c);}
	\fld{Result} node $v$ holds the substrings of lengths \texttt{t[t[v].link].len + 1} to \texttt{t[v].len}, and they share their end positions.
	The number of distinct non-empty substrings is
	\[ \textstyle\sum_{v \ge 1} \bigl(\texttt{len}(v) - \texttt{len}(\texttt{link}(v))\bigr). \]
	Occurrences of node $v$: start with 1 where \texttt{len == pos + 1} (not a clone), then add each node to its link in decreasing \texttt{len}.
	\fld{Needs} letters \texttt{FIRST}..\texttt{FIRST + K - 1}; peak memory $< 6n(K + 3)$ ints.
]\label{suffixautomaton}
\cpp[4]{Lyndon factorisation (Duval)}{lyndon.hpp}[
	\fld{Use} \texttt{vector<int> st = duval(s);}
	\fld{Result} factor $k$ is $s$ from \texttt{st[k]} to \texttt{st[k + 1] - 1}; the last one ends at $n - 1$. Letters compare as \texttt{unsigned char}.
]\label{lyndon}
\cpp[4]{Manacher}{manacher.hpp}[
	\fld{Use} \texttt{vector<int> p = manacher(s);}
	\fld{Result} \texttt{p[2*i]} gives the maximum length at \texttt{s[i]}; \texttt{p[2*i + 1]} gives it between \texttt{s[i]} and \texttt{s[i + 1]} (0 if none).
	\fld{Needs} nothing. An empty string returns an empty vector.
]\label{manacher}
\cpp[4]{Eertree}{eertree.hpp}[
	\fld{Use} \texttt{Eertree et; for(char c: s) v = et.add(c);}
	\fld{Result} \texttt{add} returns the longest palindromic suffix node so far.
	\texttt{t[v].dep} counts $v$'s non-empty palindromic suffixes, ending here for the result.
	\texttt{t.size() - 2} distinct non-empty palindromes exist.
	For occurrences, add \texttt{cnt} to its link's \texttt{cnt} for $v = \texttt{t.size() - 1},\ldots,2$.
	\fld{Needs} letters \texttt{FIRST}..\texttt{FIRST + K - 1}; memory $< 3(n + 2)(K + 4)$ ints.
	\fld{getLink} for a palindromic suffix $v$ of $s_0..s_{i-1}$ returns its longest suffix-palindrome $u$ with $s_{i-\texttt{t[u].len}-1} = s_i$.
]\label{eertree}
