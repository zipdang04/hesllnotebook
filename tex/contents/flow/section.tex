\subsection{Flow}
\cpp[4]{Dinic with scaling + min cut}{dinic.hpp}[
	\fld{Use} \texttt{Dinic d(n + 1); int id = d.addEdge(u, v, cap); ll f = d.maxFlow(s, t);}
	\fld{Result} the flow this call adds from $s$ to $t$. Vertices $0..n$ are all usable.
	\fld{Needs} $s \ne t$ and capacities $\ge 0$, with \texttt{cap + rcap} plus the total flow below $9 \cdot 10^{18}$.
	\fld{flowOf} \texttt{flowOf(id)} is the flow on a directed edge. Matching: unit edges source--left--right--sink; a left--right edge of flow 1 is a pair.
	\fld{sideS} \texttt{sideS(v)} is true on the source side of a min cut. K\"onig: left $v$ with it false and right $v$ with it true are a min vertex cover.
]\label{dinic}
\cpp[4]{MCMF}{mcmf.hpp}[
	\fld{Use} \texttt{MCMF mf(n + 1); mf.addEdge(u, v, cap, cost); auto [f, c] = mf.maxFlow(s, t);}
	\fld{Result} returns cheapest \texttt{\{flow, cost\}} for $\min(\texttt{limit}, \text{max flow})$ units. Exact flow needs \texttt{f == limit}.
	\fld{Needs} allows negative costs, no negative cycle at $\texttt{cap} > 0$. Total cost $< 9 \cdot 10^{18}$; else use \texttt{\_\_int128 cost} in \texttt{maxFlow}.
	\fld{flowOf} flow on the edge whose id \texttt{addEdge} returned.
	\fld{Models} \S\ref{flowmodels}.
]\label{mcmf}
\cpp[4]{Min-cost circulation (cost scaling)}{costscaling.hpp}[
	\fld{Use} \texttt{MinCostCirc mc(n + 1); int id = mc.addEdge(u, v, lo, hi, c); bool ok = mc.solve();}
	\fld{Result} \texttt{ok} is false if infeasible. \texttt{mc.total} (\texttt{\_\_int128}) is the min cost.
	\fld{Needs} $\texttt{lo} \le \texttt{hi}$, any signs, and one \texttt{solve} per object. With $N$ the constructor size, $4N(N + 1) \max |c| < 9 \cdot 10^{18}$. Uses Dinic (\S\ref{dinic}).
	\fld{flowOf} \texttt{mc.flowOf(id)} is the flow on the edge. \texttt{mc.pot} are optimal duals: $c + \texttt{pot[u]} - \texttt{pot[v]} \ge 0$ on every arc with room.
	\fld{Models} \S\ref{flowmodels}
]\label{costscaling}
\cpp[4]{Global min cut (Stoer--Wagner)}{stoerwagner.hpp}[
	\fld{Use} \texttt{auto [cut, side] = globalMinCut(g);} with \texttt{g[u]} $= \{v, w\}$, each edge in both lists.
	\fld{Result} the minimum total weight of edges between \texttt{side} and the rest; \texttt{side} lists the vertices of one part.
	\fld{Needs} $w \ge 0$ and $n \ge 2$ ($n = 1$ gives \texttt{\{oo, \{\}\}}). Multi-edges add up; self-loops are ignored.
]\label{stoerwagner}

\cpp[4]{Gomory--Hu tree}{gomoryhu.hpp}[
	\fld{Use} \texttt{vector<array<ll, 3>> t = gomoryHu(g);} with \texttt{g[u]} $= \{v, w\}$, each edge in both lists.
	\fld{Result} $n - 1$ edges \texttt{\{u, v, c\}}. The min cut between any $x, y$ is the smallest $c$ on the tree path from $x$ to $y$. Uses Dinic (\S\ref{dinic}).
	\fld{Needs} $w \ge 0$.
]\label{gomoryhu}

\subsection{Matching}
\cpp[4]{Hungarian}{hungarian.hpp}[
	\fld{Use} \texttt{vector<vector<ll>> a(n + 1, vector<ll>(m + 1)); auto [cost, match] = hungarian(a);}
	\fld{Result} the minimum of $\sum_i a[i][\texttt{match[i]}]$ over injective \texttt{match}; row $i$ takes column \texttt{match[i]}.
	\fld{Needs} $n \le m$, every row of size $m + 1$ as in \texttt{Use}. Maximum weight: negate \texttt{a}. A forbidden pair: cost $10^{12}$, then check the result.
]\label{hungarian}
