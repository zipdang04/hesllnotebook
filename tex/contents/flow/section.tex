\subsection{Flow}
\cpp[3]{Dinic with scaling + min cut}{dinitzwscaling.hpp}[
	Maximum flow from $s$ to $t$ in a directed graph, with capacity scaling.
	\fld{Usage} \texttt{Dinic d(n)}; \texttt{addEdge(u, v, c)} returns the edge id; \texttt{d.maxflow(s, t)} returns the flow value. Afterwards \texttt{cap(id)} is the residual capacity of an edge, and the vertices with \texttt{level[v] < 0} form the $s$ side of a minimum cut. Capacities are at most \texttt{oo} $= 2^{60}$ (use \texttt{oo} for infinite edges).
	\fld{Time} $O(V E \log U)$, $U$ = largest capacity.
	\fld{Note} the total flow must stay below $9.2 \cdot 10^{18}$. \texttt{maxflow} changes the capacities; copy the object to run it again.
]
\cpp[3]{MCMF}{mcmf.hpp}[
	Minimum-cost flow from $s$ to $t$ by successive shortest paths with a queue-based Bellman--Ford; costs may be negative but the graph must have no negative cycle.
	\fld{Usage} \texttt{MCMF<> mf(n)}; \texttt{addEdge(u, v, cap, cost)} returns the edge id; \texttt{mf.run(s, t, limit)} returns \texttt{\{flow, cost\}}, where the flow is the smaller of \texttt{limit} and the maximum flow, sent at minimum cost. Without \texttt{limit} it is a min-cost max-flow. Afterwards \texttt{flow(id)} is the flow on an edge. Use \texttt{MCMF<\_\_int128>} if the total cost can overflow \texttt{ll}.
	\fld{Time} $O(F V E)$ worst case for $F$ augmentations, much faster in practice.

]
\cpp[3]{Min-cost circulation (cost scaling)}{costscaling.hpp}[
	Minimum-cost circulation with bounds $l \le f \le r$ on every edge and arbitrary costs, by cost-scaling push-relabel.
	\fld{Usage} \texttt{MinCostCirculation mc(n)}; \texttt{addEdge(u, v, l, r, c)} returns the edge id; \texttt{mc.run()} returns \texttt{\{feasible, cost\}}, and the cost is an \texttt{\_\_int128}. Afterwards \texttt{flow(id)} is the flow on an edge and \texttt{p} holds the dual potentials. For an $s$--$t$ flow of exactly $F$, add the edge $t \to s$ with bounds $[F, F]$ and cost $0$. For supplies $b_v$, add a vertex $X$ and the edges $X \to v$ with $[b_v, b_v]$ if $b_v > 0$, or $v \to X$ with $[-b_v, -b_v]$ if $b_v < 0$.
	\fld{Time} $O(V^2 E \log(V C))$, independent of the capacities.
	\fld{Note} costs are scaled by $n + 1$, so keep $(n+1)^2 \max|c| \lesssim 10^{18}$ and the total capacity below $9.2 \cdot 10^{18}$. Tested with $|c| \le 10^9$ and capacities up to $10^{12}$.

]
\entry{Flow models}
\small\tightdisplay
\topic{Bounds $[l, r]$}
Edge capacity $r - l$; the excess at each vertex is
\[ d_v = \sum_{\text{in}} l - \sum_{\text{out}} l \]
Add $S \to v$ with cap $d_v$ ($d_v > 0$) and $v \to T$ with cap $-d_v$ ($d_v < 0$).
\begin{facts}
\item Feasible iff $\text{maxflow}(S,T) = \sum_{d_v > 0} d_v$; the flow on an edge is $l + f$.
\item With $s, t$: add $t \to s$ cap $\infty$. Max flow $= \text{maxflow}(s,t)$ on the residual (keep $t \to s$).
\item Min flow $= f_{t \to s} - \text{maxflow}(t,s)$, computed without $t \to s$.
\item Min cost: \S1.1.3.
\end{facts}
\topic{Vertex capacity}
Split $v$: $v_{\text{in}} \to v_{\text{out}}$ with cap $c_v$.
\topic{Closure}
$S \to v$ cap $p_v > 0$, $v \to T$ cap $-p_v$; if $u$ needs $v$: $u \to v$ cap $\infty$. The best closure is the $S$ side of the min cut:
\[ \max = \sum_{p_v > 0} p_v - \text{mincut} \]
\topic{Bipartite}
With max matching $M$:
\[ \text{min vertex cover} = M, \qquad \text{max independent set} = |V| - M \]
\begin{facts}
\item K\"onig: cover $= (L \setminus Z) \cup (R \cap Z)$, $Z = S$ side.
\item Hall: $L$ is matchable iff $|N(X)| \ge |X|$ for all $X \subseteq L$.
\end{facts}
\topic{DAG}
\[ \text{min disjoint path cover} = n - M \quad (\text{edges } u_{\text{out}} v_{\text{in}}) \]
On the transitive closure: min chain cover $=$ max antichain (Dilworth).

\cpp[3]{Global min cut (Stoer--Wagner)}{stoerwagner.hpp}[
	Global minimum cut of an undirected graph with non-negative weights, by Stoer and Wagner.
	\fld{Usage} \texttt{StoerWagner sw(n)}; \texttt{addEdge(u, v, w)} with $w \ge 0$ (self-loops are ignored); \texttt{sw.run()} (call once) returns the cut weight and the vertices of one side. For $n = 1$ it returns $\{0, \{\}\}$.
	\fld{Time} $O(V^3)$.
]
\cpp[3]{Gomory--Hu tree}{gomoryhu.hpp}[
	Gomory--Hu tree: the minimum cut between every pair of vertices of an undirected graph, from $n-1$ max flows (Gusfield). Uses Dinic (\S1.1.1).
	\fld{Usage} \texttt{GomoryHu t(n)}; \texttt{addEdge(u, v, w)} with $w \ge 0$; \texttt{t.run()} (call once). The tree has the edges $(i, \texttt{par[i]})$ of weight \texttt{cut[i]} for $i \ge 1$. The minimum $u$--$v$ cut is the smallest \texttt{cut} on the tree path between $u$ and $v$.
	\fld{Time} $n-1$ calls of Dinic.
	\fld{Note} the sum of all weights must stay below $10^{18}$.
]

\subsection{Matching}
\cpp[3]{Hopcroft--Karp + min vertex cover}{hopcroftkarp.hpp}[
	Maximum matching in a bipartite graph, and a minimum vertex cover.
	\fld{Usage} \texttt{HopcroftKarp hk(n, m)} with $n$ left and $m$ right vertices; \texttt{addEdge(u, v)} for left $u \in [0, n)$ and right $v \in [0, m)$; \texttt{hk.run()} returns the number of new pairs, so calling it again after more \texttt{addEdge} calls extends the matching. Afterwards \texttt{matchL[u]} and \texttt{matchR[v]} are the partners, or $-1$ if free. \texttt{cover()} returns a minimum vertex cover \texttt{\{left, right\}} (K\"onig); its complement is a maximum independent set.
	\fld{Time} $O(E \sqrt V)$.
]
\cpp[3]{Hungarian (min-cost assignment)}{hungarian.hpp}[
	Minimum-cost assignment of $n$ rows to distinct columns of an $n \times m$ cost matrix, $n \le m$.
	\fld{Usage} \texttt{Hungarian h(n, m)}; fill \texttt{h.c[i][j]} (negative costs are fine); \texttt{h.run()} returns the total cost and a vector \texttt{match} with \texttt{match[i]} the column of row $i$. \texttt{run} keeps \texttt{c}. For a maximum weight, negate the costs and the result.
	\fld{Time} $O(n^2 m)$.
	\fld{Note} exact in \texttt{ll} for $|c| \le 10^9$ and $n \le 500$.

]
\cpp[3]{General matching (blossom)}{blossom.hpp}[
	Maximum matching in a general (non-bipartite) graph, by Edmonds' blossom algorithm.
	\fld{Usage} \texttt{Blossom b(n)}; \texttt{addEdge(u, v)} adds an undirected edge, vertices $0 \dots n-1$; \texttt{b.run()} (call once) returns the size of a maximum matching. Afterwards \texttt{mate[v]} is the partner of $v$, or $n$ if $v$ is unmatched.
	\fld{Time} $O(V^3)$.
]
