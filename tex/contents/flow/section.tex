\entry{Flow models}
\small\tightdisplay
\topic{Bounds $[l, r]$}
Edge capacity $r - l$; the excess at each vertex is
\[ d_v = \sum_{\text{in}} l - \sum_{\text{out}} l \]
Add $S \to v$ with cap $d_v$ ($d_v > 0$) and $v \to T$ with cap $-d_v$ ($d_v < 0$).
\begin{facts}
\item Feasible iff $\text{maxflow}(S,T) = \sum_{d_v > 0} d_v$; the flow on an edge is $l + f$.
\item With $s, t$: add $t \to s$ cap $\infty$. Max flow $= \text{maxflow}(s,t)$ on the residual (keep $t \to s$).
\item Min flow $= f_{t \to s} - \text{maxflow}(t,s)$, computed without $t \to s$.
\item Min cost: \S1.1.3.
\end{facts}
\topic{Vertex capacity}
Split $v$: $v_{\text{in}} \to v_{\text{out}}$ with cap $c_v$.
\topic{Closure}
$S \to v$ cap $p_v > 0$, $v \to T$ cap $-p_v$; if $u$ needs $v$: $u \to v$ cap $\infty$. The best closure is the $S$ side of the min cut:
\[ \max = \sum_{p_v > 0} p_v - \text{mincut} \]
\topic{Bipartite}
With max matching $M$:
\[ \text{min vertex cover} = M, \qquad \text{max independent set} = |V| - M \]
\begin{facts}
\item K\"onig: cover $= (L \setminus Z) \cup (R \cap Z)$, $Z = S$ side.
\item Hall: $L$ is matchable iff $|N(X)| \ge |X|$ for all $X \subseteq L$.
\end{facts}
\topic{DAG}
\[ \text{min disjoint path cover} = n - M \quad (\text{edges } u_{\text{out}} v_{\text{in}}) \]
On the transitive closure: min chain cover $=$ max antichain (Dilworth).
