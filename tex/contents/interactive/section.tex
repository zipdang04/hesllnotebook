\subsection{Codes}
\entry{Linear codes: Reed--Solomon, GF(2), stuck bits}\label{linearcodes}
{\footnotesize\tightdisplay
\topic{Reed--Solomon: erased symbols}
\textit{Problem.} Pass 1 holds a message $m=(m_0,\ldots,m_{k-1})$ of $k$ numbers and may send up to $N$ symbols. The judge erases up to $E$ of them; each survivor keeps its index $i$. Pass 2 sees the survivors $(i,c_i)$ and outputs $m$. Send $n$ symbols $c_0,\ldots,c_{n-1}$ with $k+E\le n\le N$.
\par\textit{Send.} Pick a prime $p$ with $p\ge n$, $p>\max m_j$ and $p<2^{30}$ (\texttt{Mint}). Work in $\mathbb F_p$ with nodes $\alpha_i=i$:
\[ c_i=f(\alpha_i),\qquad f(t)=\sum_{j=0}^{k-1}m_j t^j. \]
\textit{Receive.} Put the nodes of any $k$ survivors in \texttt{x} and their values in \texttt{y}; \texttt{interpolate(x, y)} (\S\ref{lagrange}) returns $m$. Not \texttt{interpolateAt}: it needs the nodes $0,\ldots,n-1$ in order.
\flowfig{
  \node[note,align=center] (m) at (0,0) {$2+3t$\\over $\mathbb F_7$};
  \node[note,align=center] (c) at (1.90,0) {$t:\quad 0\quad1\quad2\quad3$\\$c:\quad 2\quad5\quad1\quad4$};
  \node[note,align=center] (r) at (4.80,0) {keep $t=1,3$\\$m=(2,3)$};
  \path (m) edge (c) (c) edge (r);
}
\textit{Errors.} If the judge may also change up to $e$ values (positions unknown) and erase up to $s$: interpolating all $u$ survivors detects 1 to $u-k$ wrong values, since a coefficient at index $k$ or higher is then nonzero. Correcting needs
\[ 2e+s\le n-k \]
and a decoder that is not in this notebook.
\topic{Random GF(2) code: erased bits}
\textit{Problem.} Pass 1 holds a message $m=(m_1,\ldots,m_k)$ of $k$ bits and may send up to $N$ bits. The judge erases up to $E$ of them; each survivor keeps its index. Pass 2 outputs $m$; a failure chance below $2^{-t}$ is allowed. Send $n=N$ bits; it works when $n\ge k+E+t$.
\par\textit{Send.} From a shared fixed seed, generate a random $n\times k$ bit matrix $G$. Send $c=Gm$ modulo 2. Example: with rows \texttt{10, 01, 11}, the message \texttt{10} is sent as \texttt{101}.
\par\textit{Receive.} Let $S$ be the surviving positions. Solve $G_S m=c_S$ ($G_S$, $c_S$: the rows and bits at $S$) with \texttt{solveXor(a, b, k)} (\S\ref{gauss}) on the surviving rows and bits; accept only a unique solution (the result has size 1). \texttt{BitRow} needs $k\le4\dsep096$.
\par\begin{tabular}{@{}lll@{}}
\toprule
Keep rows $S$ & Solve $G_S m=c_S$ & Recover $m$ \\
\midrule
$1,3$ (bits \texttt{1,1}) & $m_1=1$, $m_1+m_2=1$ & \texttt{10} \\
\bottomrule
\end{tabular}\par
\textit{Limits.} No solution means some surviving bit is wrong. If $|S|=k+t$ and the erasures do not depend on $G$,
\[ \Pr[\operatorname{rank}G_S<k]\le (2^k-1)2^{-|S|}<2^{-t}. \]
\topic{Stuck bits: store the syndrome}
\textit{Problem.} Pass 1 holds a message $m=(m_1,\ldots,m_k)$ of $k$ bits and writes it into $n$ cells $x=(x_1,\ldots,x_n)$. Some cells, the set $S$, are stuck at values $b$; only pass 1 knows $S$ and $b$. Pass 2 reads all $n$ cells and outputs $m$.
\par\textit{Send.} Both passes generate the same random $k\times n$ bit matrix $H$ from a fixed seed; all arithmetic is modulo 2. Set $x_S=b$ and solve, over the other cells $F$,
\[ H_F x_F=m+H_S b \]
with \texttt{solveXor} (\S\ref{gauss}); any solution works. An empty result means this message cannot be written. \texttt{BitRow} needs $|F|\le4\dsep096$.
\par\textit{Receive.} Output $m=Hx$.
\par\textit{Example.} $H=(1,1,1)$, cell 1 stuck at 1, $m=0$: write \texttt{110}; pass 2 gets $1+1+0=0$.
\par\textit{Limits.} If $F$ does not depend on $H$ and $|F|=k+t$, some message cannot be written with probability below $2^{-t}$. About $n-|S|$ message bits fit.
}
\entry{Bit-level codes: self-delimiting, VT, pairs, checksum}\label{bitcodes}
{\footnotesize\tightdisplay
\topic{Unary}
\textit{Problem.} Pass 1 writes several numbers into one bit string; pass 2 must split them apart. For small $v\ge0$, e.g.\ run lengths.
\par\textit{Send.} $v$ ones, then a zero: $v+1$ bits.
\[ 0\mapsto\texttt{0}\qquad 3\mapsto\texttt{1110} \]
\textit{Receive.} Count the ones up to the first zero. A missing zero signals a cut.
\topic{Elias-gamma}
\textit{Problem.} The same, for integers $v\ge1$ of widely varying size.
\par\textit{Send.} With $z=\lfloor\log_2 v\rfloor$: $z$ zeros, then $v$ in binary ($z+1$ bits), $2z+1$ bits in all. For $v\ge0$, encode $v+1$.
\par\begin{tabular}{@{}lll@{}}
$z$ zeros & binary $v$ ($z+1$ bits) & length $2z+1$ \\
\texttt{00} & \texttt{101} ($v=5$) & 5 bits \\
\end{tabular}\par
\textit{Receive.} Count the leading zeros $z$, then read $z+1$ bits from the first one.
\topic{Padding to a fixed block}
\textit{Problem.} The channel stores blocks of a fixed size and the payload is shorter; pass 2 must find where the payload ends, even when it ends in zeros.
\par\textit{Send.} The length as Elias-gamma $\gamma(\mathit{len}+1)$, the payload, then zeros. Check that header and payload fit.
\flowfig{
  \node[note,draw,minimum width=2.45cm] (a) at (0,0) {\strut $\gamma(\mathit{len}+1)$};
  \node[note,draw,minimum width=1.4cm,anchor=west] (b) at (a.east) {\strut payload};
  \node[note,draw,minimum width=1.6cm,anchor=west] (c) at (b.east) {\strut zero padding};
  \node[note,below] at (a.south) {\texttt{011} ($\mathit{len}=2$)};
  \node[note,below] at (b.south) {\texttt{00}};
  \node[note,below] at (c.south) {\texttt{0}\,$\cdots$};
}
\textit{Receive.} Decode $\mathit{len}$, then read exactly $\mathit{len}$ bits, including any trailing zeros.
\topic{Varshamov--Tenengolts: one insertion or deletion}
\textit{Problem.} Pass 1 sends a word $x$ of known length $n$. The judge deletes or inserts one bit, and flips none. Pass 2 receives $y$ and outputs $x$.
\par\textit{Send.} Position $i$ (from 1) has weight $i$. Send only words with $R(x)=a$; $a=0$ gives the most words, at least $2^n/(n+1)$, where
\[ R(x)=\Big(\sum_i i x_i\Big)\bmod(n+1) \qquad (i\text{ starts at }1). \]
Reserve positions $1,2,4,\ldots\le n$ for checks. Set them to zero and fill the other $n-\lfloor\log_2 n\rfloor-1$ slots with data. Let $d=(a-R(x))\bmod(n+1)$ in $0,\ldots,n$. Put bit $j$ of $d$ at position $2^j$.
\par\begin{tabular}{@{}l*{8}{c}@{}}
Position & 1 & 2 & 3 & 4 & 5 & 6 & 7 & 8 \\
Bit & $d_0$ & $d_1$ & data & $d_2$ & data & data & data & $d_3$ \\
\end{tabular}\par
\textit{Receive.} At length $n$, keep $y$. At length $n+1$, try deleting each bit and keep any result with residue $a$ (all give the same word).
\par At length $n-1$, count the $w$ ones in $y$. Take $\Delta=(a-R(y))\bmod(n+1)$ in $0,\ldots,n$.
\par\begin{tabular}{@{}ll@{}}
$\Delta\le w$ & Insert 0 with $\Delta$ ones to its right. \\
$\Delta>w$ & Insert 1 with $\Delta-w-1$ zeros to its left. \\
\end{tabular}\par
Example: $n=3$, $a=0$, received \texttt{11}: $w=2$, $\Delta=1\le w$, so insert a 0 with one 1 to its right: \texttt{1\underline{0}1}.
\topic{Complementary pairs}
\textit{Problem.} The judge flips bits in one direction only ($0\to1$, or $1\to0$). Pass 2 must know which bits were hit, so that each hit becomes an erasure.
\par\textit{Send.} $0$ as \texttt{01} and $1$ as \texttt{10}; keep pair boundaries aligned.
\par\textit{Receive.} Read each pair:
\par\begin{tabular}{@{}lll@{}}
\toprule
Flips in a pair & Receive & Action \\
\midrule
1 & \texttt{00} or \texttt{11} & Mark erased (\S\ref{linearcodes}). \\
2 & other valid pair & Error goes undetected. \\
\bottomrule
\end{tabular}\par
\textit{Limits.} The second row needs flips in both directions; with one-way flips every hit is seen.
\topic{Two checksums}
\textit{Problem.} Pass 1 sends $n$ numbers $x_i$ and two sums that arrive intact. The judge changes one symbol, position unknown. Pass 2 finds and repairs it.
\par\textit{Send.} Choose a prime $p>n$, $p>\max x_i$, and index from 1:
\[ S_0=\sum_{i=1}^{n}x_i,\qquad S_1=\sum_{i=1}^{n}i x_i. \]
\textit{Receive.} Recompute both sums from the received symbols; $\Delta_0,\Delta_1$ are recomputed minus sent, modulo $p$.
\par\begin{tabular}{@{}lll@{}}
\toprule
Change & Locate & Repair \\
\midrule
One symbol & $j=\Delta_1/\Delta_0$ & $x_j\gets x_j-\Delta_0$ \\
$2\to5$ in $\mathbb F_{11}$ & $j=9/3=3$ & $5-3=2$ \\
\bottomrule
\end{tabular}\par
\textit{Limits.} Multiple changes can give a false position. If $\Delta_0=0\ne\Delta_1$, report an unlocatable error. Three or more changes can leave both sums unchanged.
\topic{Hash a candidate}
\textit{Problem.} Pass 2 can rebuild a few candidate originals. Pass 1 sends a short trusted digest so that pass 2 can pick the true one.
\par\textit{Send.} The length and \texttt{StrHash(s).get(1, sz(s))} (\S\ref{strhash}); characters must have codes 1 to 127. Use the same base in both passes (\S\ref{multipasscheck}).
\par\textit{Receive.} Keep the candidate whose length and digest match.
\par\textit{Limits.} A match is probabilistic and cannot locate an error.
}
\cpp[4]{Hamming SECDED}{hamming.hpp}[
	\fld{Use} \texttt{auto w = hammingEncode(x); ll y = hammingDecode(w);}
	\fld{Result} \texttt{y == x} after at most one flipped bit of \texttt{w}, and $-1$ after exactly two (single-error correction, double-error detection).
	\fld{Needs} $0 \le x < 2^{57}$. Three or more flips can give a wrong \texttt{y} silently.
]\label{hamming}

\subsection{Encoding}
\cpp[4]{Tree canonical form (AHU)}{ahu.hpp}[
	\fld{Use} \texttt{AHU t; vector<int> id = t.rooted(g, r);}
	\fld{Result} \texttt{id[u] == id[v]} iff the subtrees of $u$ and $v$ under root $r$ are isomorphic; ids run over $1..\texttt{sz(t.ids)}$.
	\fld{Needs} a tree on $1..n$, $n \ge 1$, each edge in both lists. \texttt{t = AHU()} clears the table.
	\fld{unrooted} \texttt{t.unrooted(g) == t.unrooted(h)} iff $g$ and $h$ are isomorphic as unrooted trees.
]\label{ahu}
\cpp[4]{Rank / unrank}{rankunrank.hpp}[
	\fld{Use} \texttt{Bigint r = rankSubset(a); a = unrankSubset(n, k, r);}
	\fld{Result} \texttt{a} is a sorted $k$-subset of $1..n$ and $0 \le r < \binom{n}{k}$, which is \texttt{rankBinom(n, k)}.
	\fld{Needs} Bigint (\S\ref{bigint}). Multisets: $\max a_i + k - 1 \le 2^{31}-1$.
	\fld{rankMultiset} \texttt{unrankMultiset(n, k, r)} inverts it on nondecreasing $k$-tuples of $1..n$; there are $\binom{n+k-1}{k}$.
	\fld{rankPerm} \texttt{unrankPerm(n, r)} inverts it on the $n!$ permutations.
]\label{rankunrank}
