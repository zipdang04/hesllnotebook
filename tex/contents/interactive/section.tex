\cpp[4]{Hamming SECDED}{hamming.hpp}[
	\fld{Use} \texttt{auto w = hammingEncode(x); ll y = hammingDecode(w);}
	\fld{Result} \texttt{y == x} after at most one flipped bit of \texttt{w}, and $-1$ after exactly two (single-error correction, double-error detection).
	\fld{Needs} $0 \le x < 2^{57}$. Three or more flips can give a wrong \texttt{y} silently.
]\label{hamming}

\subsection{Encoding}
\cpp[4]{Tree canonical form (AHU)}{ahu.hpp}[
	\fld{Use} \texttt{AHU t; vector<int> id = t.rooted(g, r);}
	\fld{Result} \texttt{id[u] == id[v]} iff the subtrees of $u$ and $v$ under root $r$ are isomorphic; ids run over $1..\texttt{sz(t.ids)}$.
	\fld{Needs} a tree on $1..n$, $n \ge 1$, each edge in both lists. \texttt{t = AHU()} clears the table.
	\fld{unrooted} \texttt{t.unrooted(g) == t.unrooted(h)} iff $g$ and $h$ are isomorphic as unrooted trees.
]\label{ahu}
