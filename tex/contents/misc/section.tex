\cpp[4]{DP opt: D\&C, Knuth, Aliens}{dpopt.hpp}[
	\textbf{QI}: $C(a,c) + C(b,d) \le C(a,d) + C(b,c)$ for all $a \le b \le c \le d$.
	\textbf{D\&C}: $dp_i[j] = \min_{k<j} dp_{i-1}[k] + C(k, j)$ (group $(k, j{]})$) needs the leftmost optimal $k$ non-decreasing in $j$; QI suffices. Layers: $dp_1[j] = C(0, j)$, then \texttt{dnc(i, n, i-1, n-1, dp$_{i-1}$, dp$_i$, C)} for $i = 2..K$; $O(Kn\log n)$. \texttt{prev[k] + C} must not overflow.
	\textbf{Knuth}: $dp[l][r] = \min_{l \le k < r} dp[l][k] + dp[k+1][r] + C(l, r)$, any $dp[i][i]$; needs QI and $C(b,c) \le C(a,d)$ ($a \le b \le c \le d$) $\Rightarrow opt[l][r-1] \le opt[l][r] \le opt[l+1][r]$. $O(n^2)$ time, $12n^2$ bytes. Optimal BST: dummies $0..n$, split $k$ = root key $k+1$, $dp[i][i] = q_i$.
	\textbf{Aliens}: $f(k)$ = min with exactly $k$ items, integer and convex (e.g.\ $k$ groups with a QI cost). \texttt{solve($\lambda$)} adds $\lambda$ per item and returns \{min, \emph{smallest} count among the minimisers\}; result $=$ value $- \lambda^* k$ for the smallest $\lambda^* \in [lo, hi]$ with count $\le k$. Needs $lo \le f(k-1) - f(k)$, $hi \ge f(k) - f(k+1)$. The count at $\lambda^*$ may be $< k$ (ties): only the value is recovered. ll: $|\text{value}| + \max(|lo|, |hi|) \cdot n < 9.2 \cdot 10^{18}$.
]
