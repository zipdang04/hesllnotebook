\subsection{Number theory}
\cpp[3]{Modint}{modint.hpp}[\texttt{MOD} prime, $< 2^{30}$; NTT, FPS, BM need $998244353$.]
\cpp{Bigint}{bigint.hpp}[Base 10. \texttt{*}: NTT (\S6.3.2) if both $\ge 64$ digits, needs $|a|+|b| \le 2^{22}+1$ digits ($2\cdot10^6 \times 2\cdot10^6$: 0.8\,s). \texttt{/}, \texttt{\%} by \texttt{ll} $0<|b|\le10^{17}$: $O(|a|)$; by bigint: $O(|q|\,|b|)$, $10^4/5\cdot10^3$ digits 0.5\,s, $2\cdot10^4/10^4$ 1.9\,s. C++ signs: $q$ truncates toward 0, $r$ has the sign of $a$.]
\cpp[4]{Ext-gcd, CRT}{crt.hpp}[
	Any $a, b$; \texttt{crt} needs $m, n \le 4\cdot 10^{18}$, $\mathrm{lcm}(m,n) < 2^{63}$; no solution $\to \{-1,-1\}$. Many congruences: fold from \texttt{\{0, 1\}}.
]
\cpp[3]{Sieve (factorable)}{sieve.hpp}[$O(n)$; memory $\approx 4n$ bytes ($n=5\cdot10^8$: 2.2\,GB, 1.7\,s).]
\cpp[3]{Primality, factorization}{primes.hpp}[\texttt{isPrime}: deterministic for all $n < 2^{64}$ (7 bases). \texttt{factor}: $n \ge 1$, prime factors with multiplicity, \emph{unsorted}; $\approx O(n^{1/4})$ expected.]
\cpp[4]{Discrete log / sqrt}{dlog.hpp}[$1\le m\le 10^9$ (tested). \texttt{dlog}: $a, b \ge 0$ (not necessarily coprime), $0^0=1$, $-1$ if none; \texttt{dsqrt}: $p$ prime, other root $p-x$. Primitive root: $g$ is one iff $g^{(p-1)/q}\ne1$ for every prime $q\mid p-1$ (text only, no code).]
\cpp[4]{Floor sum}{floorsum.hpp}[
	$0 \le n < 2^{32}$, $1 \le m < 2^{32}$, any $a, b$ ($|a|, |b| \le 4\cdot 10^{18}$); exact if the answer fits in \texttt{ll}.
]

\subsection{Linear algebra}
\cpp[4]{Gauss--Jordan}{gauss.hpp}[
	Prime modulus. \texttt{rref(a, m, \&det)}: pivots in the first $m$ columns, the rest are carried along (\texttt{solve}: {$[A|b]$}; \texttt{inverse}: {$[A|I]$}); \texttt{det} valid for square $A$. \texttt{solveXor}: $b$ in bit $m$ of each row, $m+1 \le$ \texttt{MAXB}; $O(n^2m/64)$.
]
\cpp[4]{Simplex}{simplex.hpp}[
	Max $c^T x$ s.t.\ $Ax \le b$, $x \ge 0$; $\ge$ rows: negate; $=$: two rows; free $x = x^+ - x^-$. Returns $-\infty$ (infeasible, $x$ untouched), $\infty$ (unbounded, $x$ feasible) or the max with an optimal $x$. \texttt{eps} $= 10^{-9}$ is absolute: tested $|A|, |c| \le 100$, $|b| \le 10^5$; scale the data otherwise.
]

\subsection{Polynomials \& transforms}
\cpp[3]{FFT}{fft.hpp}[\texttt{conv}: exact after rounding if $(\sum a_i^2+\sum b_i^2)\log_2 n < 9\cdot10^{14}$. \texttt{convMod}: $a_i, b_i \in {[0, M)}$, tested $M \le 10^9+7$, $|a|=|b|=2^{19}$ (0.25\,s); balanced digits keep rounding error $\le 0.2$ in pure \texttt{double} (plain {$[0,\sqrt M)$} digits fail).]
\cpp{NTT}{ntt.hpp}[
	$p = c \cdot 2^k + 1$, $g$ a primitive root mod $p$ (3 for $998244353 = 119 \cdot 2^{23} + 1$): $\omega = g^c$ is a primitive $2^k$-th root of unity, so lengths up to $2^k$. Other primes: \S9.2.
]
\cpp[4]{FWHT (xor/and/or), SOS}{fwht.hpp}[$|a|=2^k$, $O(n\log n)$. The or/and transforms are subset/superset sums (SOS DP); replacing $+$ by $\max$/$\min$ (\texttt{sos}) gives SOS over max/min.]
\cpp[5]{FPS: inv, log, exp, div}{fps.hpp}[
	Mod $998244353$ only; results have exactly $n$ terms ($a$ may be shorter/longer). \texttt{inv}: $a_0 \ne 0$; \texttt{log}: $a_0 = 1$; \texttt{exp}: $a_0 = 0$; \texttt{divMod}: $b$ has no leading zero, $q, r$ untrimmed. $n = 5\cdot10^5$: exp 4\,s, others $< 2$\,s.
]
\cpp[5]{Berlekamp--Massey + Bostan--Mori}{bm.hpp}[
	Mod $998244353$ only. \texttt{berlekampMassey}: shortest $c$, $O(n^2)$; an order-$d$ recurrence needs $2d$ terms; all zeros $\to$ \texttt{\{\}}. \texttt{bostanMori}: $[x^k]P/Q$, $Q_0 \ne 0$, any $|P|$, $O(m\log m\log k)$. \texttt{kthTerm}: $a_k$ from $a_0..a_{d-1}$ and $c$ ($d=0 \to 0$), replaces Kitamasa; guess: \texttt{kthTerm(s, berlekampMassey(s), k)}. $d=10^5$, $k=10^{18}$: 3.5\,s.
]
\cpp[4]{Lagrange interpolation}{lagrange.hpp}[$O(n)$ at $x=0..n-1$ (prime $p\ge n$); \texttt{interpolate}: distinct $x$, $O(n^2)$, then Horner.]
