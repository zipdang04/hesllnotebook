\subsection{Number theory}
\cpp[4]{Modint}{modint.hpp}[
	\fld{Use} \texttt{Mint<1\cppsep000\cppsep000\cppsep007> a = 3; a = a * b / c; cout << a.pow(k) << a.inv();}
	\fld{Result} \texttt{a.v} holds the residue, $0 \le \texttt{v} < \texttt{MOD}$. An \texttt{ll} converts to \texttt{Mint}, so \texttt{2 * a} and \texttt{a == 0} work.
	\fld{Needs} a prime \texttt{MOD}; divisors and arguments of \texttt{inv} are nonzero; \texttt{pow(n)} takes $n \ge 0$.
	\fld{MintN} is \texttt{Mint<998\cppsep244\cppsep353>}, the type NTT, FPS and the linear recurrence use. Other moduli can be used next to it.
]\label{modint}
\cpp[4]{Bigint}{bigint.hpp}[
	\fld{Use} \texttt{Bigint a("123"), b = -5; a = a * b + a / b; cout << a << (a < b);}
	\fld{Result} \texttt{a.neg} is the sign and \texttt{a.d} holds base-$10^9$ limbs, lowest first; zero has no limbs.
	\fld{Needs} a product below $8 \cdot 10^6$ digits; $b \ne 0$ for \texttt{/} and \texttt{\%}. Uses NTT (\S\ref{ntt}) on \texttt{MintN}.
	\fld{divMod} \texttt{a.divMod(b)} is $\{q, r\}$ with $a = qb + r$ and $|r| < |b|$; $q$ is truncated and $r$ has the sign of $a$, as in C++.
]\label{bigint}
\cpp[4]{Sieve (factorable)}{sieve.hpp}[
	\fld{Use} \texttt{Sieve sv(n);} then \texttt{sv.primes}, \texttt{sv.minDiv[x]}, \texttt{sv.factorize(x)}.
	\fld{Result} \texttt{factorize(x)} lists the prime factors of $x$ with multiplicity; $x = 1$ gives none.
	\fld{Needs} $n \le 10^8$ in 512 MB: 4 bytes per number. Add $\varphi$ or $\mu$ in the inner loop (\S\ref{mathnt}).
]\label{sieve}
\cpp[4]{Ext-gcd, CRT}{crt.hpp}[
	\fld{Use} \texttt{ll g = extGcd(a, b, x, y); auto [r, l] = crt(a, m, b, n);}
	\fld{Result} \texttt{extGcd}: $ax + by = g = \gcd(a, b)$; if $g = 1$, $x \equiv a^{-1} \pmod b$, maybe negative: reduce $x \bmod b$, then add $b$ only if that is negative.
	\fld{crt} Returns $r \equiv a \pmod m$, $r \equiv b \pmod n$, $0 \le r < l = \mathrm{lcm}(m, n)$; \texttt{\{-1, -1\}} if none.
	\endgraf\noindent For many congruences, fold from \texttt{\{0, 1\}} and stop at $-1$.
]\label{crt}
\cpp[4]{Miller--Rabin}{millerrabin.hpp}[
	\fld{Use} \texttt{bool p = isPrime(n);}
	\fld{Result} exact for every \texttt{ll} $n$, false below 2: the 7 bases are proven for all $n < 2^{64}$.
	\fld{mulMod} returns $a \cdot b \bmod m$.
	\fld{powMod} returns $a^n \bmod m$.
]\label{millerrabin}
\cpp[4]{Pollard rho}{pollardrho.hpp}[
	\fld{Use} \texttt{vector<ll> f = factorize(n);}
	\fld{Result} prime factors with multiplicity, unsorted; $n = 1$ gives none.
	\fld{Needs} $1 \le n < 2^{62}$. Uses \texttt{isPrime} (\S\ref{millerrabin}).
	\fld{findDivisor} a divisor $d$ of a composite $n$, $1 < d < n$, not always prime. It never returns on a prime $n$.
]\label{pollardrho}
\cpp[4]{Discrete log, discrete sqrt}{dlog.hpp}[
	\fld{Use} \texttt{ll k = discreteLog(a, b, m);}
	\fld{Result} the smallest $k \ge 0$ with $a^k \equiv b \pmod m$, or $-1$; $a$ and $m$ need not be coprime.
	\fld{Needs} Miller--Rabin (\S\ref{millerrabin}) for \texttt{mulMod} and \texttt{powMod}.
	\fld{discreteSqrt} $x$ with $x^2 \equiv a \pmod p$ for a prime $p$, or $-1$; the other root is $p - x$.
]\label{dlog}
\cpp[4]{Floor sum}{floorsum.hpp}[
	\fld{Use} \texttt{ll s = floorSum(n, m, a, b);}
	\fld{Result} the sum of $(a i + b) \bmod m$ over the same $i$ is $\sum_i (a i + b) - m s$.
	\fld{Needs} $n \ge 0$ and $m \ge 1$. Every intermediate fits in \texttt{ll} when
	\[ (n + 1)\,m < 9 \cdot 10^{18}, \quad n^2 (|a| + 2m) + 2n (|b| + m) < 18 \cdot 10^{18} \cdot m. \]
]\label{floorsum}

\subsection{Linear algebra}
\cpp[4]{Gauss--Jordan}{gauss.hpp}[
	\fld{Use} \texttt{auto [rank, det] = gauss(a, c);} Uses Modint (\S\ref{modint}).
	\fld{Result} reduces $n \times m$ \texttt{Mint<MOD>} matrix \texttt{a} on columns $0..c-1$. Set $c = m$ for rank, $c = n = m$ for determinant (0 if singular).
	\fld{solveLinear} solves $Ax = b$, $n \ge 1$: empty if impossible, else \texttt{s[0]} solves it and \texttt{s[1..]} span the kernel ($Ax = 0$).
	\fld{matInverse} sets \texttt{a} to $A^{-1}$; singular: false, \texttt{a} undefined.
	\fld{solveXor} solves over $\mathrm{GF}(2)$ likewise, with \texttt{bitset} rows; use width $\max(4\dsep097, m + 1)$.
]\label{gauss}
{\footnotesize\tightdisplay
\topic{Real numbers}
For a system over \texttt{double} (probabilities, expected values), copy \texttt{gauss} and make these edits.
\begin{facts}
	\item Drop \texttt{template<int MOD>} and write \texttt{double} for \texttt{Mint<MOD>}; \texttt{x.inv()} becomes \texttt{1 / x}.
	\item Pivot on the row with the largest $|a_{r,\mathrm{col}}|$, not on the first nonzero one.
	\item Treat $|x| < 10^{-9}$ as zero, in the pivot search, the no-solution test and the \texttt{piv} scan of \texttt{solveLinear}.
\end{facts}
}
\cpp[4]{Simplex}{simplex.hpp}[
	\fld{Use} \texttt{Simplex lp(A, b, c);} then test \texttt{lp.val} against \texttt{-INFINITY} and \texttt{INFINITY}.
	\fld{Result} \texttt{lp.x} reaches \texttt{lp.val}. It is empty for \texttt{-INFINITY} and some feasible point for \texttt{INFINITY}.
	\fld{Needs} negate $c$ for a minimum, negate a row for $\ge$, and give two rows for $=$.
	It runs fastest when $b \ge 0$, so that $x = 0$ is feasible.
]\label{simplex}

\subsection{Polynomials \& transforms}
\cpp[4]{FFT}{fft.hpp}[
	\fld{Use} \texttt{vector<ll> c = convMod(a, b, mod);}
	\fld{Result} $|a| + |b| - 1$ terms in ${[0, \texttt{mod})}$, empty when an input is empty:
	\[ c_k = \sum_i a_i b_{k-i} \bmod \texttt{mod} \]
	\fld{Needs} $0 \le a_i, b_i < \texttt{mod} \le 2^{30}$ and $|a| + |b| - 1 \le 2^{20}$.
	\fld{conv} in \texttt{double}; exact after \texttt{llround} while $(\sum a_i^2 + \sum b_i^2) \log_2(|a| + |b|) < 9 \cdot 10^{14}$.
	\fld{fft} forward only. Inverse: \texttt{fft(a); reverse(a.begin() + 1, a.end());} divide by $n$.
]\label{fft}
\cpp[4]{NTT}{ntt.hpp}[
	\fld{Use} \texttt{vector<MintN> c = convNtt(a, b);}
	\fld{Result} $|a| + |b| - 1$ terms, empty when an input is empty:
	\[ c_k = \sum_i a_i b_{k-i} \]
	\fld{Needs} \texttt{MintN} modulus $c2^k+1$, primitive root 3, $|a|+|b|-1 \le 2^k$. Uses Modint (\S\ref{modint}); $k=23$ for $998\dsep244\dsep353$.
	\fld{ntt} several products of one size: pad to a power of two $n \ge |a| + |b| - 1$, transform each input once, multiply pointwise, then \texttt{ntt(c, true)}.
]\label{ntt}
\cpp[4]{FWHT}{fwht.hpp}[
	\fld{Use} \texttt{vector<MintN> c = convBit(a, b, '\textasciicircum');} or \texttt{'\&'}, \texttt{'|'}.
	\fld{Result} for the operation $\circ$ given, sizes padded to a power of two:
	\[ c_k = \sum_{i \circ j = k} a_i b_j. \]
	\fld{Needs} \texttt{MOD} $\ne 2$ for \texttt{'\textasciicircum'}. Uses Modint (\S\ref{modint}).
	\fld{fwht} transforms in place, size a power of two; \texttt{fwht(a, op, true)} inverts it.
]\label{fwht}
{\footnotesize\tightdisplay
\topic{Subset sums (SOS)}
To get $\sum_{T \subseteq S} a_T$ for every mask $S$, call \texttt{fwht(a, '|')}; for supersets, \texttt{'\&'}.
\texttt{fwht(a, '|', true)} is Möbius inversion over subsets. For \texttt{ll} values, copy the loop with \texttt{ll} in place of \texttt{Mint<MOD>}.
\topic{Max and min convolution}
For $c_k = \sum_{\max(i, j) = k} a_i b_j$, multiply prefix sums $A$, $B$ pointwise and take differences:
\[ A_k = \sum_{i \le k} a_i, \qquad C_k = A_k B_k, \qquad c_k = C_k - C_{k-1}. \]
For $\min(i, j) = k$, use suffix sums $A_k = \sum_{i \ge k} a_i$ and $c_k = C_k - C_{k+1}$.
}
\cpp[4]{FPS: inv, log, exp, div}{fps.hpp}[
	\fld{Use} \texttt{vector<MintN> g = fpsInv(a, n);}
	\fld{Result} the first $n$ terms of $1/a$, $\log a$ or $\exp a$; \texttt{a} may be shorter or longer than $n$.
	\fld{Needs} $1 \le n \le 2^{22}$ and $|a| - |b| < 2^{22}$; $a_0 \ne 0$ for \texttt{fpsInv}, $a_0 = 1$ for \texttt{fpsLog}, $a_0 = 0$ for \texttt{fpsExp}. Uses NTT (\S\ref{ntt}).
	\fld{polyDiv} \texttt{q = polyDiv(a, b)} has $|a| - |b| + 1$ terms, none if $|a| < |b|$. The remainder is the first $|b| - 1$ terms of $a - bq$.
]\label{fps}
\cpp[4]{Berlekamp--Massey + Bostan--Mori}{linearrec.hpp}[
	\fld{Use} \texttt{auto c = berlekampMassey(s); MintN x = kthTerm(s, c, k);}
	\fld{Result} \texttt{c} is the shortest recurrence of \texttt{s}, of order $d = |c|$, and \texttt{x} is $s_k$:
	\[ s_i = \sum_{j=0}^{d-1} c_j \, s_{i-1-j} \quad (i \ge d). \]
	\fld{Needs} $2d$ terms find an order-$d$ recurrence; \texttt{kthTerm} needs $|s| \ge d$, $0 \le k \le 10^{18}$, $2d + 1 \le 2^{23}$. Uses NTT (\S\ref{ntt}).
	\fld{kthTerm} For another prime, retype \texttt{MintN}; then \texttt{kthTerm} needs $d \le 31$.
]\label{linearrec}
\cpp[4]{Lagrange interpolation}{lagrange.hpp}[
	\fld{Use} \texttt{vector<MintN> c = interpolate(x, y);}
	\fld{Result} the coefficients of the unique $f$ with $f(x_i) = y_i$.
	\fld{Needs} distinct $x_i$ (a repeat gives a wrong $f$ silently); $1 \le n \le \texttt{MOD}$ for \texttt{interpolateAt}. Uses Modint (\S\ref{modint}).
	\fld{interpolateAt} returns $f(t)$ for any \texttt{ll} $t$ from the values $y_i = f(i)$, without the coefficients.
]\label{lagrange}
{\footnotesize\tightdisplay
\topic{Formula}
Evaluate the interpolating polynomial by hand, or check a coefficient, with
\[ f(t) = \sum_i y_i \prod_{j \ne i} \frac{t - x_j}{x_i - x_j}. \]
}

\subsection{Numerical}
\cpp[4]{Simpson integration}{simpson.hpp}[
	\fld{Use} \texttt{double s = simpson(a, b, [\&](double x)\{ return f(x); \});}
	\fld{Result} $\int_a^b f(x)\,dx$ with estimated error below \texttt{eps} for a smooth $f$; $a > b$ gives the negative.
	\fld{Needs} split at kinks, jumps, peaks narrower than $(b - a) / 4$, cancelling terms ($16x^6 - 65x^4$ on $[0, 1]$ gives $-10.7083$, not $-75/7$).
	\fld{eps} absolute threshold on the error estimate, not a bound; scale it up when $\max |f|\,(b - a) > 10^7$.
]\label{simpson}
