\subsection{Segment trees}
\cpp[4]{Lazy segment tree}{lazyseg.hpp}[
	\fld{Use} \texttt{vector<LazySeg::Node> a(n + 1); a[i] = \{x, 1\}; LazySeg seg(a);}
	\fld{Result} \texttt{seg.get(l, r).sum} returns $a_l + \dots + a_r$; \texttt{seg.update(l, r, \{b, c\})} sets $a_i \gets b a_i + c$ on $[l, r]$, modulo 998\dsep244\dsep353 (\texttt{MintN}, \S\ref{modint}).
	\fld{Needs} \texttt{combine} and \texttt{apply} have identities \texttt{Node()} and \texttt{Tag()}.
	\fld{apply} composes \texttt{t} after \texttt{id}'s tag. For ``set'', reserve an unused \texttt{Tag} value and return at once on it.
	\fld{combine} \texttt{combine(x, y)} has \texttt{x} left of \texttt{y}; it need not commute.
]\label{lazyseg}
\cpp[4]{Persistent segment tree}{perseg.hpp}[
	\fld{Use} \texttt{vector<PerSeg::Node> a(n + 1); PerSeg s(a); int v = s.update(0, p, \{x\});}
	\fld{Result} \texttt{update(v, p, x)} returns the id of a new version: version \texttt{v} with $a_p = x$. Old versions never change.
	\fld{Needs} \texttt{a} is not empty; \texttt{Node()} and \texttt{combine} as in Lazy segment tree (\S\ref{lazyseg}).
	\fld{get} \texttt{s.get(v, l, r).sum} is $a_l + \dots + a_r$ in version \texttt{v}.
	\fld{update} Point add is \texttt{update(v, p, \{get(v, p, p).sum + x\})}. Range add: the next topic.
]\label{perseg}
{\footnotesize\tightdisplay
\topic{Range add with never-pushed tags}
To add $x$ on $[l, r]$ in a new version, give every node a tag \texttt{add} that covers its whole range and is never pushed.
\begin{facts}
	\item \texttt{sum} of a node includes its own \texttt{add}, not its ancestors'. Let $\mathit{len} = \mathit{hi} - \mathit{lo} + 1$ be the node's length.
	\item \texttt{update(id, lo, hi, l, r, x)} returns \texttt{id} on no overlap and copies every other visited node. Full cover: \texttt{sum += x * len}, \texttt{add += x}, stop. Partial: recurse into both children, then
	\[ \mathit{sum} = \mathit{sum}_L + \mathit{sum}_R + \mathit{add} \cdot \mathit{len}. \]
	\item \texttt{get} carries $c$, the sum of the ancestors' \texttt{add} (0 at the root), and passes $c + \mathit{add}$ to the children. Full cover returns
	\[ \mathit{sum} + c \cdot \mathit{len}. \]
	\item Store a returned child id in a local before writing it into the arena: the arena may move during the call.
\end{facts}
\topic{Assign or affine on a range}
When tags do not commute (assign, $x \to bx + c$), their order matters, so push them down with copying.
\begin{facts}
	\item Before entering the children of a node, copy both children, apply the node's tag to the copies, point the node at them and clear its tag (\texttt{push} of \S\ref{lazyseg}).
	\item \texttt{get} pushes too, so a query also makes $O(\log n)$ nodes. The pushed node may stay shared: its contents do not change.
\end{facts}
}
\cpp[4]{Li Chao tree}{lichao.hpp}[
	\fld{Use} \texttt{LiChao lc(pts); lc.addLine(a, b); ll y = lc.get(x);}
	\fld{Result} \texttt{get(x)} is the minimum of $ax + b$ over the lines covering $x$, or \texttt{LiChao::oo} if none does. For a maximum, add $(-a, -b)$, negate.
	\fld{Needs} \texttt{pts} (not empty) holds every $x$ asked. For each line (negated for a max) and each $x$ of \texttt{pts} it covers, $ax$ and $ax + b$ fit in \texttt{ll} and $ax + b < \texttt{oo}$.
	\fld{addSegment} \texttt{addSegment(a, b, l, r)} adds $ax + b$ only for $l \le x \le r$.
]\label{lichao}

\subsection{Balanced trees}
\cpp[4]{Implicit treap}{treap.hpp}[
	\fld{Use} \texttt{Treap tr; tr.insert(tr.t[tr.root].size + 1, x);}
	\fld{Result} \texttt{get(l, r).sum} sums $[l, r]$ mod 998\dsep244\dsep353 (\texttt{MintN}, \S\ref{modint}). \texttt{reverse(l, r)} reverses it; \texttt{update(l, r, \{b, c\})} sets $a_i \gets b a_i + c$.
	\fld{Needs} Lazy segment tree EDIT rules (\S\ref{lazyseg}); \texttt{pull} sets \texttt{size}. For non-commutative sums, swap a reversed sum at each \texttt{swap(l, r)}.
	\fld{insert} \texttt{insert(pos, x)} makes $x$ element \texttt{pos}, $\texttt{pos} \le \texttt{size} + 1$ (the Use line appends); \texttt{erase(pos)} removes it.
	\fld{split} \texttt{split(u, k)} cuts root \texttt{u} after $k$ elements; \texttt{merge(a, b)} joins.
]\label{treap}
\cpp[4]{Link-cut tree}{linkcut.hpp}[
	\fld{Use} \texttt{LinkCut lct(n + 1); lct.update(u, a\_u); lct.link(u, v);}
	\fld{Result} \texttt{lct.get(u, v).sum} sums the values on the path $u$ to $v$; \texttt{update(u, v, \{x\})} adds $x$ to each, \texttt{update(u, x)} sets $u$'s.
	\fld{Needs} \texttt{link} joins two trees, \texttt{cut} removes an edge; path calls and \texttt{lca} need $u$, $v$ in one tree. Edit as \S\ref{lazyseg}; \texttt{pull} sets \texttt{size}.
	\fld{lca} \texttt{lca(u, v)} uses the current root: \texttt{makeRoot(r)} and path calls $(u, v)$ root at $r$, $u$; \texttt{link(u, v)} keeps $v$'s; \texttt{cut(u, v)} roots at $u$, $v$.
	\fld{flip} Non-commutative \texttt{sum}: \texttt{pull} sets a reversed sum, \texttt{flip} swaps it.
]\label{linkcut}

\subsection{Other}
\cpp[4]{Rollback DSU}{rollbackdsu.hpp}[
	\fld{Use} \texttt{RollbackDsu dsu(n + 1); int t = dsu.time(); dsu.unite(u, v); dsu.rollback(t);}
	\fld{Result} \texttt{time()} is the number of joins kept, so $1..n$ has $n - \texttt{time()}$ components. \texttt{size(u)} is the size of the component of $u$.
	\fld{unite} returns false, and records nothing, if $u$ and $v$ are already together.
	\fld{rollback} \texttt{rollback(t)} undoes every join made after \texttt{t = time()}; a later \texttt{t} does nothing.
]\label{rollbackdsu}
\cpp[4]{pb\_ds}{pbds.hpp}[
	\fld{Use} \texttt{OrderedSet<int> s; OrderedMap<int, ll> om; gp\_hash\_table<ll, int, CustomHash> m;}
	\fld{Result} \texttt{s.order\_of\_key(x)} counts keys $< x$: $x$'s rank, even if absent. \texttt{om} is a \texttt{map} with the same two calls; \texttt{find\_by\_order} gives a pair iterator.
	\fld{Needs} distinct keys; for a multiset use \texttt{OrderedSet<pii>} with unique \texttt{.se}. Pack pair keys of \texttt{m} into one \texttt{ll}.
	\fld{find\_by\_order} \texttt{s.find\_by\_order(k)} points at rank $k$, or \texttt{end()} if $k \ge \texttt{s.size()}$; test before \texttt{*}.
	\fld{CustomHash} also serves \texttt{unordered\_map<ll, int, CustomHash>}; its seed is random per run.
]\label{pbds}
\cpp[4]{Mo's algorithm}{mo.hpp}[
	\fld{Use} \texttt{vector<ll> ans = moQueries(q, add, del, [\&]\{ return cur; \});} with \texttt{q[i] = \{l, r\}}.
	\fld{Result} \texttt{ans[i]} is \texttt{get()} on the window $a_l, \dots, a_r$ of \texttt{q[i]}.
	\fld{Needs} $1 \le l \le r$ in every query, and \texttt{del} undoes \texttt{add} on the same side.
	\fld{del} is called only on a window that still contains \texttt{q[i]}'s range, so it never empties the window.
]\label{mo}
