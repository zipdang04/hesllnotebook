\subsection{Segment trees}
\cpp[4]{Lazy segment tree}{lazyseg.hpp}[
	\fld{Use} \texttt{vector<LazySeg::Node> a(n + 1); a[i] = \{x, 1\}; LazySeg seg(a);}
	\fld{Result} \texttt{seg.get(l, r).sum} returns $a_l + \dots + a_r$; \texttt{seg.update(l, r, \{b, c\})} sets $a_i \gets b a_i + c$ on $[l, r]$, modulo 998244353 (\texttt{MintN}, \S\ref{modint}).
	\fld{Needs} \texttt{combine} and \texttt{apply} have identities \texttt{Node()} and \texttt{Tag()}.
	\fld{apply} composes \texttt{t} after \texttt{id}'s tag. For ``set'', reserve an unused \texttt{Tag} value and return at once on it.
	\fld{combine} \texttt{combine(x, y)} has \texttt{x} left of \texttt{y}; it need not commute.
]\label{lazyseg}
\cpp[3]{Persistent segment tree}{ps_seg.hpp}
\cpp[4]{Li Chao tree}{lichao.hpp}[
	\fld{Use} \texttt{LiChao lc(pts); lc.addLine(a, b); ll y = lc.get(x);}
	\fld{Result} \texttt{get(x)} is the minimum of $ax + b$ over the lines covering $x$, or \texttt{LiChao::oo} if none does. For the maximum, add $(-a, -b)$ and negate.
	\fld{Needs} \texttt{pts} holds every $x$ ever asked (any order, duplicates fine, not empty); any other $x$ gives a wrong answer silently. For each line and each covered $x$ of \texttt{pts}, $ax$ and $ax + b$ fit in \texttt{ll} and $ax + b < \texttt{oo}$.
	\fld{addSegment} \texttt{addSegment(a, b, l, r)} adds $ax + b$ only for $l \le x \le r$; it does nothing if no point of \texttt{pts} is in $[l, r]$.
]\label{lichao}

\subsection{Balanced trees}
\cpp[4]{Implicit treap}{treap.hpp}[
	\fld{Use} \texttt{Treap tr; tr.insert(1, x);} appends $x$.
	\fld{Result} \texttt{get(l, r).sum} sums $[l, r]$ mod 998244353 (\texttt{MintN}, \S\ref{modint}). \texttt{reverse(l, r)} reverses it; \texttt{update(l, r, \{b, c\})} sets $a_i \gets b a_i + c$.
	\fld{Needs} Lazy segment tree EDIT rules (\S\ref{lazyseg}); \texttt{pull} sets \texttt{size}. For non-commutative sums, swap a reversed sum at each \texttt{swap(l, r)}.
	\fld{insert} \texttt{insert(pos, x)} inserts $x$ at $1 \le \texttt{pos} \le \texttt{size} + 1$; \texttt{erase(pos)} removes it.
	\fld{split} \texttt{split(u, k)} cuts root \texttt{u} after $k$; \texttt{merge(a, b)} joins roots.
]\label{treap}

\subsection{Other}
\cpp[4]{Rollback DSU}{rollbackdsu.hpp}[
	\fld{Use} \texttt{RollbackDsu dsu(n + 1); int t = dsu.time(); dsu.unite(u, v); dsu.rollback(t);}
	\fld{Result} \texttt{time()} is the number of joins kept, so $1..n$ has $n - \texttt{time()}$ components. \texttt{size(u)} is the size of the component of $u$.
	\fld{unite} returns false, and records nothing, if $u$ and $v$ are already together.
	\fld{rollback} \texttt{rollback(t)} undoes every join made after \texttt{t = time()}; a later \texttt{t} does nothing.
]\label{rollbackdsu}
\cpp[4]{pb\_ds}{pbds.hpp}[
	\fld{Use} \texttt{OrderedSet<int> s; gp\_hash\_table<ll,int,CustomHash> mp;}
	\fld{Result} \texttt{s.order\_of\_key(x)} counts keys $< x$: $x$'s rank, even if absent.
	\fld{Needs} distinct keys; for a multiset use \texttt{OrderedSet<pii>} with unique \texttt{.se}. Pack pair keys of \texttt{mp} into one \texttt{ll}.
	\fld{find\_by\_order} \texttt{s.find\_by\_order(k)} points at rank $k$, or \texttt{end()} if $k \ge \texttt{s.size()}$; test before \texttt{*}.
	\fld{CustomHash} also serves \texttt{unordered\_map<ll, int, CustomHash>}; its seed is random per run.
]\label{pbds}
